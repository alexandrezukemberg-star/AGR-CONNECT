'use strict'; const $=id=>document.getElementById(id);let client=null,me=null,peer=null,channel=null,signingUp=false; const config=window.AGR_CONNECT_CONFIG||{}; if(config.url&&config.publishableKey&&window.supabase){client=window.supabase.createClient(config.url,config.publishableKey)} const say=(id,message)=>$(id).textContent=message; function err(e){return e?.message||'Não foi possível concluir a operação.'} function toggle(){signingUp=!signingUp;$('signupFields').classList.toggle('hidden',!signingUp);$('authTitle').textContent=signingUp?'Criar conta':'Entrar';$('authSubmit').textContent=signingUp?'Cadastrar':'Entrar';$('toggleAuth').textContent=signingUp?'Já tenho uma conta':'Criar uma conta';$('displayName').required=signingUp;$('username').required=signingUp;$('password').autocomplete=signingUp?'new-password':'current-password'} $('toggleAuth').onclick=toggle; $('authForm').onsubmit=async e=>{e.preventDefault();if(!client)return say('authNotice','Conecte primeiro seu projeto Supabase em config.js.');const email=$('email').value.trim(),password=$('password').value;try{if(signingUp){const username=$('username').value.trim().toLowerCase();const {error}=await client.auth.signUp({email,password,options:{data:{username,display_name:$('displayName').value.trim()}}});if(error)throw error;say('authNotice','Confira seu e-mail e confirme o cadastro.
